程序结构有三要素：顺序（从上到下逐行执行）、条件（走哪条分支）、循环（重复做几遍）。 本页把条件语句与循环语句各做几组对照演示，左边的代码就是右边结果的真实来源。

顺序结构按书写顺序一行一行执行，默认状态
条件结构if / else if / else、switch、三元运算符
循环结构for、while、do…while、循环嵌套

一、条件结构

// 用 if / else if / else 判断成绩等级
function gradeOf(score) {
    if (score >= 90) {
        return "优秀";
    } else if (score >= 80) {   // 上面不成立才轮到这句
        return "良好";
    } else if (score >= 60) {
        return "及格";
    } else {
        return "不及格";
    }
}
// 用 switch 判断星期几：值等于哪个 case 就走哪个
function weekOf(n) {
    switch (n) {
        case 1: return "星期一";
        case 2: return "星期二";
        case 3: return "星期三";
        case 4: return "星期四";
        case 5: return "星期五";
        case 6:
        case 7: return "周末";       // case 6 没有 break，会接着走 case 7
        default: return "不是 1-7 的数字";
    }
}
// 三元运算符：条件 ? 成立时的值 : 不成立时的值
var n = 7;
var oddEven = n % 2 === 0 ? "偶数" : "奇数";

// 条件也可以用逻辑运算符组合
var year = 2025;
var leap = (year % 4 === 0 && year % 100 !== 0) || year % 400 === 0;

二、循环结构

// for：次数已知，初始化 / 条件 / 步长 三句话在一行
var sum = 0;
for (var i = 1; i <= 100; i++) {
    sum += i;
}
// while：先判断条件，成立才执行循环体
var t = 5;
var word = "";
while (t > 0) {
    word += t + " ";
    t--;
}
word += "发射！";
// do…while：先执行一次，再判断条件
var num = 10;
var times = 0;
do {
    times++;      // 条件一开始就不成立，但循环体仍执行了 1 次
    num++;
} while (num < 10);

三、循环嵌套：九九乘法表

外层循环管行，内层循环管列，内层的条件 j <= i 让每一行少一格，最终排成三角形。

// 外层 i 控制行，内层 j 控制列
var html = "";
for (var i = 1; i <= 9; i++) {
    for (var j = 1; j <= i; j++) {
        html += "<span>" + j + "×" + i + "=" + (i * j) + "</span>";
    }
}

四、条件 + 循环：1-100 里有多少个"数7"

这就是 15-1 那个数7游戏背后的判断逻辑：i % 7 === 0 是 7 的倍数， String(i).indexOf("7") !== -1 是数字中含 7，两个条件用 || 连起来。 15-1 里留空的 30 个格子，就是下面这些数字。 

// 循环负责"逐个检查"，条件负责"要不要留空"
var count = 0;
var hit = [];

for (var i = 1; i <= 100; i++) {
    if (i % 7 === 0 || String(i).indexOf("7") !== -1) {
        count++;
        hit.push(i);   // push 把符合规则的数放进数组
    }
}

五、循环里的两个跳转

// break：直接结束整个循环
var first = 0;
for (var k = 1; k <= 100; k++) {
    if (k % 7 === 0) {
        first = k;    // 找到第一个 7 的倍数就退出
        break;
    }
}

// continue：跳过这一次，继续下一次
var odd = "";
for (var m = 1; m <= 10; m++) {
    if (m % 2 === 0) {
        continue;      // 偶数跳过，只拼奇数
    }
    odd += m + " ";
}
