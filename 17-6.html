主流程：async / await 写法

// 每个步骤都是一个返回 Promise 的函数：到点了就 resolve
function task(name, el, ms) {
    return new Promise(function (resolve) {
        setState(el, "running");
        setTimeout(function () {
            setState(el, "done");
            resolve();               // 到点才算"完成"
        }, ms);
    });
}

async function makeDumplings() {
    // 线上、线下同时开始，谁先完成就继续（另一个还在路上）
    await Promise.race([task("线上买", online, 6000),
                        task("线下买", offline, 8000)]);

    await task("取原料", pick, 2000);

    // 和面、和馅同时进行，两个都完成才往下走
    await Promise.all([task("和面", dough, 4000),
                       task("和馅", filling, 3000)]);

    await task("包饺子", wrap, 5000);
    await task("煮饺子", boil, 4000);
    await task("吃饺子", eat, 2000);
}

    })
    .catch(function (err) {
        console.log("哪一步失败都会到这里：" + err.message);
    });

为什么用 race 和 all

Promise.race([A, B]) 返回一个新的 Promise，只要 A 或 B 有一个完成它就完成 —— 对应"线上线下哪边先到都行"。Promise.all([A, B]) 则是两个都完成才完成， 而且总耗时等于较慢的那个 —— 对应"面和馅都齐了才能包"。 所以并行比串行快：面和馅串着做要 7 秒，同时做只要 4 秒。
